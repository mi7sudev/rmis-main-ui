<script>
const TOTAL = 6;
let cur = 2; // matches the mock: first two done
const titles = [
  "Apply for this Role.","Choose your Position.","Tell us your Story.",
  "Your Education.","Upload your Documents.","Review & Submit."
];

function render(){
  document.querySelectorAll('.step').forEach((s,i)=>{
    s.classList.toggle('done', i < cur && !(i===cur));
    s.classList.toggle('current', i === cur);
    const d = document.getElementById('dot'+i);
    if(i < cur){ d.innerHTML='✓'; } else { d.innerHTML=''; }
  });
  document.querySelectorAll('.panel').forEach(p=>{
    p.classList.toggle('active', +p.dataset.panel === cur);
  });
  document.getElementById('headTitle').textContent = titles[cur];
}
function next(){ if(cur < TOTAL-1){ cur++; render(); } }
function prev(){ if(cur > 0){ cur--; render(); } }
document.querySelectorAll('.step').forEach(s=>{
  s.addEventListener('click',()=>{ cur = +s.dataset.step; render(); });
});

/* dropzones */
function fakeDrop(el, label){
  el.classList.add('filled');
  el.querySelector('.dz-t').textContent = label + '_final.pdf ✓';
  el.querySelector('.dz-s').textContent = '1.2 MB · Click to replace';
}
function fakeUpload(el, name){
  el.textContent = '✓ ' + name + ' attached';
  el.style.color = '#1fa04b';
}

/* review + submit */
document.getElementById('appForm').addEventListener('submit', e=>{
  e.preventDefault();
  const rows = [...document.querySelectorAll('[data-field]')]
    .filter(f=>f.value.trim())
    .map(f=>`<div class="review-row"><span class="k">${f.dataset.field}</span><span class="v">${f.value}</span></div>`)
    .join('');
  document.getElementById('reviewList').innerHTML = rows ||
    '<p style="color:var(--muted);font-size:13.5px;padding:10px 4px">No details filled yet — you can still submit, or go back and complete your profile.</p>';
  document.getElementById('appForm').style.display='none';
  document.querySelector('.card-head').style.display='none';
  document.getElementById('success').classList.add('show');
  // mark all steps done
  cur = TOTAL; // keep stepper visual
  document.querySelectorAll('.step').forEach((s,i)=>{
    s.classList.remove('current'); s.classList.add('done');
    document.getElementById('dot'+i).innerHTML='✓';
  });
});

render();
</script>
